cmake_minimum_required(VERSION 3.16)
project(receiver_replay C ASM)
include(../models/assets.cmake)
add_library(receiver_core ../main/decoder.c "${assets}")
target_include_directories(receiver_core PUBLIC ../main)
target_link_libraries(receiver_core PUBLIC m)
target_compile_options(receiver_core PRIVATE -Wall -Wextra)
add_executable(replay replay.c)
target_link_libraries(replay PRIVATE receiver_core)
if(UNIX)
    add_executable(receiverctl receiverctl.c)
endif()
enable_testing()
add_executable(test_capture test_capture.c)
target_link_libraries(test_capture PRIVATE receiver_core)
add_test(NAME capture COMMAND test_capture "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/queue-admission.tsv")
add_executable(test_history test_history.c ../main/history.c ../main/cpr.c)
target_link_libraries(test_history PRIVATE receiver_core)
add_test(NAME history COMMAND test_history)
add_executable(test_cpr test_cpr.c ../main/cpr.c)
target_include_directories(test_cpr PRIVATE ../main)
target_link_libraries(test_cpr PRIVATE m)
add_test(NAME cpr COMMAND test_cpr)
add_executable(test_tuning test_tuning.c ../main/tuning.c)
target_include_directories(test_tuning PRIVATE ../main)
add_test(NAME tuning COMMAND test_tuning)
add_executable(test_gps_nmea test_gps_nmea.c ../techo/gps_nmea.c)
target_include_directories(test_gps_nmea PRIVATE ../techo)
target_compile_options(test_gps_nmea PRIVATE -Wall -Wextra)
add_test(NAME gps_nmea COMMAND test_gps_nmea)
add_test(NAME recordings COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/check.sh" $<TARGET_FILE:replay>)
find_package(ZLIB REQUIRED)
add_executable(test_tdeck test_tdeck.c ../main/tdeck_ui.c ../main/tdeck_ubx.c
    ../main/history.c ../main/cpr.c ../main/tdeck_map.c ../main/tdeck_clock.c)
target_compile_definitions(test_tdeck PRIVATE TDECK_REPLAY_HOST_TEST)
target_link_libraries(test_tdeck PRIVATE receiver_core ZLIB::ZLIB)
target_compile_options(test_tdeck PRIVATE -Wall -Wextra)
add_test(NAME tdeck_recordings COMMAND sh "${CMAKE_CURRENT_SOURCE_DIR}/check-tdeck.sh"
    $<TARGET_FILE:replay> $<TARGET_FILE:test_tdeck>)
if(UNIX)
    add_executable(test_tdeck_storage test_tdeck_storage.c ../main/observation.c ../main/tdeck_map.c)
    target_include_directories(test_tdeck_storage PRIVATE . ../main)
    target_link_libraries(test_tdeck_storage PRIVATE ZLIB::ZLIB m)
    add_test(NAME tdeck_storage_shutdown COMMAND test_tdeck_storage)
endif()
